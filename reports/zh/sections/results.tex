\subsection{主要结果}

\begin{theorem}\label{thm:main}
对表\ref{tab:results}中的每个维数 $n$，存在含 $N_n$ 个点的接吻构型，其中 $N_n$ 为第二列的下界。因此 $K(n)\ge N_n$。
\end{theorem}
